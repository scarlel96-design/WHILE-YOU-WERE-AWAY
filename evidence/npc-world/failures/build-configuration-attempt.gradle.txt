// Developer clients are muted BEFORE launch, including the title/onboarding screens.
// Only isolated development options are changed, never the user's normal .minecraft profile.
['runClient': file('run'), 'runClientSmoke': file('../../work/client-smoke'), 'runCitySmoke':file('../../work/city-smoke'),'runCityReload':file('../../work/city-smoke')].each { taskName, gameDir ->
    tasks.named(taskName).configure {
        doFirst {
            gameDir.mkdirs()
            def options = new File(gameDir, 'options.txt')
            def lines = options.exists() ? options.readLines('UTF-8') : []
            def overrides = ['soundCategory_master':'0.0','soundCategory_music':'0.0',
                'soundCategory_record':'0.0','soundCategory_weather':'0.0','soundCategory_block':'0.0',
                'soundCategory_hostile':'0.0','soundCategory_neutral':'0.0','soundCategory_player':'0.0',
                'soundCategory_ambient':'0.0','soundCategory_voice':'0.0','narrator':'0',
                'onboardAccessibility':'false','tutorialStep':'none','pauseOnLostFocus':'false']
            lines.removeAll { line -> overrides.containsKey(line.split(':', 2)[0]) }
            if (taskName != 'runClient') { lines.removeAll { it.startsWith('lang:') || it.startsWith('renderDistance:') || it.startsWith('simulationDistance:') }; lines.add('lang:ko_kr') }
            overrides.each { k,v -> lines.add("${k}:${v}") }
            options.setText(lines.join('\n')+'\n','UTF-8')
            if (!options.text.contains('soundCategory_master:0.0')) throw new GradleException('Prelaunch mute verification failed')
            logger.lifecycle("VERIFIED MUTED before launch: ${options.absolutePath}")
            if (taskName == 'runClientSmoke') file('evidence/client-smoke.txt').setText('RUNNING: awaiting client checks\n', 'UTF-8')
            if (taskName.startsWith('runCity')) file('evidence/stability/regression/'+taskName+'.txt').setText('RUNNING\n','UTF-8')
        }
        if (taskName.startsWith('runCity')) {
            doLast {
                if(!file('evidence/stability/regression/'+taskName+'.txt').getText('UTF-8').contains('PASS '+taskName))
                    throw new GradleException('City client did not finish every gate')
            }
        }
        if (taskName == 'runClientSmoke') {
            doLast {
                def evidence = file('evidence/client-smoke.txt').getText('UTF-8')
                if (!evidence.contains('PASS tracked custom entity present with camera line of sight') ||
                    !evidence.contains('PASS scene S2C starts, reaches caption phase and ends; master volume remains zero')) {
                    throw new GradleException('Client smoke did not reach every required gate; inspect runtime log')
                }
            }
        }
    }
}

// Fault-injection clients use the same prelaunch mute implementation as every other test client.
['Cut','Copy','Resume'].each { mode -> (1..2).each { repetition ->
    tasks.named("runRecovery${mode}${repetition}").configure {
        if(mode == 'Cut') ignoreExitValue = true
        doFirst {
            def gameDir=file('../../work/recovery-smoke');gameDir.mkdirs()
            def options=new File(gameDir,'options.txt')
            def lines=options.exists()?options.readLines('UTF-8'):[]
            def muted=['master','music','record','weather','block','hostile','neutral','player','ambient','voice']
            lines.removeAll { it.startsWith('soundCategory_') || it.startsWith('narrator:') || it.startsWith('onboardAccessibility:') || it.startsWith('tutorialStep:') || it.startsWith('pauseOnLostFocus:') || it.startsWith('lang:') || it.startsWith('renderDistance:') || it.startsWith('simulationDistance:') }
            muted.each { lines.add("soundCategory_${it}:0.0") }
            lines.addAll(['narrator:0','onboardAccessibility:false','tutorialStep:none','pauseOnLostFocus:false','lang:ko_kr','renderDistance:8','simulationDistance:5'])
            options.setText(lines.join('\n')+'\n','UTF-8')
            if(!muted.every { options.text.contains("soundCategory_${it}:0.0") })throw new GradleException('Recovery mute failed')
            logger.lifecycle("VERIFIED MUTED before launch: ${options.absolutePath}")
            file("evidence/stability/runRecovery${mode}${repetition}.txt").setText('RUNNING\n','UTF-8')
        }
        doLast {
            if(mode == 'Cut') logger.lifecycle("EXPECTED_FAULT_PROCESS_EXIT=${executionResult.get().exitValue}")
            if(!file("evidence/stability/runRecovery${mode}${repetition}.txt").text.contains("PASS runRecovery${mode}${repetition}"))throw new GradleException('Recovery fault gate did not finish')
        }
    }
} }

                    !evidence.contains('PASS scene S2C starts, reaches caption phase and ends; master volume remains zero')) {

    }
